let cart = JSON.parse(localStorage.getItem("cart")) || []; function updateCartCount(){ document.querySelectorAll(".cart-count").forEach(el=>{ el.innerText = cart.length; }); } function saveCart(){ localStorage.setItem("cart", JSON.stringify(cart)); updateCartCount(); } function addToCart(name, price){ cart.push({name, price}); saveCart(); alert("Added to cart!"); } function displayCart(){ let output = ""; let total = 0; cart.forEach((item, index)=>{ output += item.name + " - $" + item.price + `
`; total += item.price; }); document.getElementById("cartItems").innerHTML = output; document.getElementById("totalPrice").innerHTML = "Total: $" + total; } function removeItem(index){ cart.splice(index,1); saveCart(); displayCart(); } function checkout(){ alert("Payment Successful! Order Confirmed."); cart = []; saveCart(); displayCart(); } window.onload = function(){ updateCartCount(); }
